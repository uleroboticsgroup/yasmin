99 static std::vector<uint8_t> py_buffer_to_vector(const py::handle &value) {
100 char *data = nullptr;
101 py::ssize_t size = 0;
102
103 if (PyBytes_Check(value.ptr()) != 0) {
104 if (PyBytes_AsStringAndSize(value.ptr(), &data, &size) != 0) {
105 throw py::error_already_set();
106 }
107 } else if (PyByteArray_Check(value.ptr()) != 0) {
108 data = PyByteArray_AsString(value.ptr());
109 size = PyByteArray_Size(value.ptr());
110 } else {
111 throw std::runtime_error("Expected bytes or bytearray object");
112 }
113
114 return std::vector<uint8_t>(reinterpret_cast<uint8_t *>(data),
115 reinterpret_cast<uint8_t *>(data) + size);
116 }

202 template <typename T>
203 static std::vector<T> sequence_to_vector(const py::sequence &seq) {
204 std::vector<T> result;
205 result.reserve(static_cast<std::size_t>(py::len(seq)));
206
207 for (auto item : seq) {
208 result.push_back(py::cast<T>(item));

251 template <typename T>
252 static std::unordered_map<std::string, T>
253 dict_to_unordered_map(const py::dict &dict) {
254 std::unordered_map<std::string, T> result;
255 result.reserve(static_cast<std::size_t>(py::len(dict)));
256
257 for (auto item : dict) {
258 result.emplace(py::cast<std::string>(item.first),
259 py::cast<T>(item.second));
260 }
261
262 return result;

316 void set(const std::string &key, py::object value) {
317 py::gil_scoped_acquire gil;
318 if (py::isinstance<py::bool_>(value)) {
319 this->blackboard->set<bool>(key, value.cast<bool>());
320 return;
321 }
322
323 if (is_python_int_like(value)) {
324 this->blackboard->set<std::int64_t>(key, value.cast<std::int64_t>());
325 return;
326 }
327
328 if (py::isinstance<py::float_>(value)) {
329 this->blackboard->set<double>(key, value.cast<double>());
330 return;
331 }
332
333 if (is_python_bytes_like(value)) {
334 this->blackboard->set<std::vector<uint8_t>>(key,
335 py_buffer_to_vector(value));
336 return;
337 }
338
339 if (py::isinstance<py::str>(value)) {
340 this->blackboard->set<std::string>(key, value.cast<std::string>());
341 return;
342 }
343
344 if (yasmin::callback_signal_pyutils::is_python_callback_signal_like(
345 value)) {
346 this->blackboard->set<yasmin::CallbackSignal::SharedPtr>(
347 key,
348 yasmin::callback_signal_pyutils::cast_python_callback_signal(value));
349 return;
350 }
351
352 if (is_python_sequence_like(value)) {
353 py::sequence seq = value.cast<py::sequence>();
354
355 // Empty sequences do not carry enough type information to infer a native
356 // element type safely.
357 if (py::len(seq) == 0) {
358 this->blackboard->set<py::object>(key, value);
359 return;
360 }
361
362 // Bool must be checked before int because Python bool is also an int.
363 if (sequence_matches(seq, [](const py::handle &item) {
364 return py::isinstance<py::bool_>(item);
365 })) {
366 this->blackboard->set<BoolVector>(key, sequence_to_vector<bool>(seq));
367 return;
368 }
369
370 if (sequence_matches(seq, [](const py::handle &item) {
371 return is_python_int_like(item);
372 })) {
373 this->blackboard->set<IntVector>(key,
374 sequence_to_vector<std::int64_t>(seq));
375 return;
376 }
377
378 // Mixed int/float sequences are normalized to double.
379 if (sequence_matches(seq, [](const py::handle &item) {
380 return is_python_number_like(item);
381 })) {
382 this->blackboard->set<FloatVector>(key,
383 sequence_to_vector<double>(seq));
384 return;
385 }
386
387 if (sequence_matches(seq, [](const py::handle &item) {
388 return py::isinstance<py::str>(item);
389 })) {
390 this->blackboard->set<StringVector>(
391 key, sequence_to_vector<std::string>(seq));
392 return;
393 }
394
395 this->blackboard->set<py::object>(key, value);
396 return;
397 }
398
399 if (py::isinstance<py::dict>(value)) {
400 py::dict dict = value.cast<py::dict>();
401
402 // Only dict[str, T] with a homogeneous value type is converted into a
403 // native unordered_map.
404 if (py::len(dict) == 0 || !dict_has_only_string_keys(dict)) {
405 this->blackboard->set<py::object>(key, value);
406 return;
407 }
408
409 if (dict_values_match(dict, [](const py::handle &item) {
410 return py::isinstance<py::bool_>(item);
411 })) {
412 this->blackboard->set<BoolDict>(key, dict_to_unordered_map<bool>(dict));
413 return;
414 }
415
416 if (dict_values_match(dict, [](const py::handle &item) {
417 return is_python_int_like(item);
418 })) {
419 this->blackboard->set<IntDict>(
420 key, dict_to_unordered_map<std::int64_t>(dict));
421 return;
422 }
423
424 // Mixed int/float dictionaries are normalized to double.
425 if (dict_values_match(dict, [](const py::handle &item) {
426 return is_python_number_like(item);
427 })) {
428 this->blackboard->set<FloatDict>(key,
429 dict_to_unordered_map<double>(dict));
430 return;
431 }
432
433 if (dict_values_match(dict, [](const py::handle &item) {
434 return py::isinstance<py::str>(item);
435 })) {
436 this->blackboard->set<StringDict>(
437 key, dict_to_unordered_map<std::string>(dict));
438 return;
439 }
440
441 this->blackboard->set<py::object>(key, value);
442 return;
443 }
444
445 this->blackboard->set<py::object>(key, value);
446 }

456 py::object get(const std::string &key) const {
457 const std::string type = this->blackboard->get_type(key);
458
459 // py::object is stored directly without extra wrapping
460 if (type == demangle_type(typeid(py::object).name())) {
461 return this->blackboard->get<py::object>(key);
462 }
463
464 using Getter =
465 std::function<py::object(const Blackboard &, const std::string &)>;
466
467 static const auto *const GETTERS = []() {
468 using Map = std::unordered_map<std::string, Getter>;
469 auto *m = new Map();
470
471 auto add = [&](auto dummy) {
472 using T = decltype(dummy);
473 (*m)[demangle_type(typeid(T).name())] = [](const Blackboard &bb,
474 const std::string &k) {
475 return py::cast(bb.get<T>(k));
476 };
477 };
478
479 auto add_bytes = [&](auto dummy) {
480 using T = decltype(dummy);
481 (*m)[demangle_type(typeid(T).name())] = [](const Blackboard &bb,
482 const std::string &k) {
483 return detail::byte_vector_to_py_bytes(bb.get<T>(k));
484 };
485 };
486
487 detail::register_standard_py_types(add, add_bytes);
488
489 return m;
490 }();
491
492 auto it = GETTERS->find(type);
493 if (it != GETTERS->end()) {
494 return it->second(*this->blackboard, key);
495 }
496
497 throw std::runtime_error("Unsupported type for Python conversion: " + type);
498 }

533 static py::object raw_entry_to_pyobject(const Blackboard::RawEntry &entry) {
534 using Getter = std::function<py::object(const std::shared_ptr<void> &)>;
535
536 static const auto *const RAW_GETTERS = []() {
537 using Map = std::unordered_map<std::string, Getter>;
538 auto *m = new Map();
539
540 auto add = [&](auto dummy) {
541 using T = decltype(dummy);
542 (*m)[demangle_type(typeid(T).name())] =
543 [](const std::shared_ptr<void> &ptr) {
544 return py::cast(*std::static_pointer_cast<T>(ptr));
545 };
546 };
547
548 auto add_bytes = [&](auto dummy) {
549 using T = decltype(dummy);
550 (*m)[demangle_type(typeid(T).name())] =
551 [](const std::shared_ptr<void> &ptr) {
552 return detail::byte_vector_to_py_bytes(
553 *std::static_pointer_cast<T>(ptr));
554 };
555 };
556
557 detail::register_standard_py_types(add, add_bytes);
558
559 return m;
560 }();
561
562 auto it = RAW_GETTERS->find(entry.type_name);
563 if (it != RAW_GETTERS->end()) {
564 return it->second(entry.value);
565 }
566
567 throw std::runtime_error("Unsupported type for Python conversion: " +
568 entry.type_name);
569 }

597 py::list items() const {
598 auto keys = this->blackboard->keys();
599 auto entries = this->blackboard->get_raw_batch(keys);
600 py::list result;
601
602 for (size_t i = 0; i < keys.size(); i++) {
603 result.append(py::make_tuple(keys[i], raw_entry_to_pyobject(entries[i])));
604 }

606 return result;
