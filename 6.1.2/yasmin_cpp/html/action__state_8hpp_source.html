289 void cancel_state() override {
290
291 bool waiting_for_cancel = false;
292
293 {
294 std::lock_guard<std::mutex> lock(this->goal_handle_mutex);
295
296 if (this->goal_handle && !this->action_done_.load()) {
297 try {
298 auto callback_guard = this->callback_guard;
299 this->action_client->async_cancel_goal(
300 this->goal_handle,
301 [this, callback_guard](typename rclcpp_action::Client<
302 ActionT>::CancelResponse::SharedPtr) {
303 std::lock_guard<std::mutex> guard_lock(callback_guard->mutex);
304 if (callback_guard->alive) {
305 this->cancel_done();
306 }
307 });
308 waiting_for_cancel = true;
309 } catch (const std::exception &e) {
310 YASMIN_LOG_WARN("Failed to cancel action '%s': %s",
311 this->action_name.c_str(), e.what());
312 }
313 }
314 }
315
316 // Wait for cancellation to complete outside of the goal_handle_mutex lock
317 if (waiting_for_cancel) {
318 std::unique_lock<std::mutex> cancel_lock(this->action_cancel_mutex);
319 this->action_cancel_cond.wait_for(
320 cancel_lock, std::chrono::seconds(1),
321 [this]() { return this->cancel_done_; });
322 this->cancel_done_ = false;
323 }
324
325 // Wake up the execute() method if it's waiting
326 yasmin::State::cancel_state();
327 this->action_done_cond.notify_all();
328 }
329
336 void cancel_done() {
337 {
338 std::lock_guard<std::mutex> lock(this->action_cancel_mutex);
339 this->cancel_done_ = true;
340 }
341 this->action_cancel_cond.notify_all();
342 }

360 std::string execute(yasmin::Blackboard::SharedPtr blackboard) override {
361
362 std::unique_lock<std::mutex> lock(this->action_done_mutex);
363 int retry_count = 0;
364
365 // Wait for the action server to be available
366 YASMIN_LOG_INFO("Waiting for action '%s'", this->action_name.c_str());
367
368 const auto action_wait_timeout =
369 std::chrono::duration<int64_t, std::ratio<1>>(this->wait_timeout);
370 const auto wait_slice = std::chrono::milliseconds(100);
371 auto waited = std::chrono::milliseconds::zero();
372
373 while (!this->action_client->wait_for_action_server(wait_slice)) {
374
375 if (this->is_canceled()) {
376 return basic_outcomes::CANCEL;
377 }
378
379 if (this->wait_timeout < 0) {
380 continue;
381 }
382
383 waited += wait_slice;
384
385 if (waited < action_wait_timeout) {
386 continue;
387 }
388
389 waited = std::chrono::milliseconds::zero();
390 YASMIN_LOG_WARN("Timeout reached, action '%s' is not available",
391 this->action_name.c_str());
392 if (retry_count < this->maximum_retry) {
393 retry_count++;
394 YASMIN_LOG_WARN("Retrying to connect to action '%s' "
395 "(%d/%d)",
396 this->action_name.c_str(), retry_count,
397 this->maximum_retry);
398 } else {
399 return basic_outcomes::TIMEOUT;
400 }
401 }
402
403 if (this->is_canceled()) {
404 return basic_outcomes::CANCEL;
405 }
406
407 Goal goal = this->create_goal_handler(blackboard);
408
409 // Prepare options for sending the goal
410 const uint64_t epoch = ++this->action_epoch_;
411 auto callback_guard = this->callback_guard;
412 SendGoalOptions send_goal_options;
413 send_goal_options.goal_response_callback =
414 [this, callback_guard, epoch](GoalResponseArg goal_response) {
415 ActionState::goal_response_callback_guarded(this, callback_guard,
416 epoch, goal_response);
417 };
418
419 send_goal_options.result_callback =
420 [this, callback_guard,
421 epoch](const typename GoalHandle::WrappedResult &result) {
422 std::lock_guard<std::mutex> lock(callback_guard->mutex);
423 if (callback_guard->alive && this->action_epoch_.load() == epoch) {
424 this->result_callback(result);
425 }
426 };
427
428 if (this->feedback_handler) {
429 // blackboard is a SharedPtr captured by value, keeping the blackboard
430 // alive for the duration of the feedback callback
431 send_goal_options.feedback_callback =
432 [this, callback_guard, epoch,
433 blackboard](typename GoalHandle::SharedPtr,
434 std::shared_ptr<const Feedback> feedback) {
435 std::lock_guard<std::mutex> lock(callback_guard->mutex);
436 if (callback_guard->alive && this->action_epoch_.load() == epoch) {
437 this->feedback_handler(blackboard, feedback);
438 }
439 };
440 }
441
442 // Send the goal to the action server
443 YASMIN_LOG_INFO("Sending goal to action '%s'", this->action_name.c_str());
444 this->action_done_.store(false);
445 this->action_client->async_send_goal(goal, send_goal_options);
446
447 // Reset retry_count for the response-wait phase
448 retry_count = 0;
449
450 if (this->response_timeout > 0) {
451 // Use a single total timeout instead of a retry loop that doesn't
452 // re-send the request on each iteration
453 const auto total_timeout =
454 std::chrono::seconds(static_cast<int64_t>(this->response_timeout) *
455 (static_cast<int64_t>(this->maximum_retry) + 1));
456 if (!this->action_done_cond.wait_for(lock, total_timeout, [this]() {
457 return this->action_done_.load() || this->is_canceled();
458 })) {
459 if (this->is_canceled()) {
460 return basic_outcomes::CANCEL;
461 }
462 return basic_outcomes::TIMEOUT;
463 }
464
465 } else {
466 this->action_done_cond.wait(lock, [this]() {
467 return this->action_done_.load() || this->is_canceled();
468 });
469 }
470
471 if (this->is_canceled()) {
472 return basic_outcomes::CANCEL;
473 }
474
475 switch (this->action_status) {
476 case rclcpp_action::ResultCode::CANCELED:
477 return basic_outcomes::CANCEL;
478
479 case rclcpp_action::ResultCode::ABORTED:
480 return basic_outcomes::ABORT;
481
482 case rclcpp_action::ResultCode::SUCCEEDED:
483 if (this->result_handler) {
484 return this->result_handler(blackboard, this->action_result);
485 }
486 return basic_outcomes::SUCCEED;
487
488 default:
489 return basic_outcomes::ABORT;
490 }
491 }
