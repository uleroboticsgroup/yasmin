42class PythonCallbackHolder {
43public:
44 explicit PythonCallbackHolder(py::function callback)
45 : callback_(new py::function(std::move(callback))) {}
46
47 PythonCallbackHolder(const PythonCallbackHolder &) = delete;
48 PythonCallbackHolder &operator=(const PythonCallbackHolder &) = delete;
49
50 ~PythonCallbackHolder() {
51 try {
52 auto &internals = py::detail::get_internals();
53 (void)internals;
54 } catch (...) {
55 return;
56 }
57
58 py::gil_scoped_acquire acquire;
59 delete this->callback_;
60 this->callback_ = nullptr;
61 }
62
63 py::function &callback() { return *this->callback_; }
64
65private:
66 py::function *callback_;
67};

81inline Blackboard::SharedPtr
82convert_blackboard_from_python(const py::object &blackboard_obj) {
83 Blackboard::SharedPtr blackboard;
84
85 // Case 1: None or not provided - create new Blackboard
86 if (blackboard_obj.is_none()) {
87 blackboard = yasmin::Blackboard::make_shared();
88 }
89 // Case 2: Check if it's a BlackboardPyWrapper
90 else if (py::isinstance<yasmin::BlackboardPyWrapper>(blackboard_obj)) {
91 auto &wrapper = blackboard_obj.cast<yasmin::BlackboardPyWrapper &>();
92 // Get the shared pointer directly instead of copying
93 blackboard = wrapper.get_cpp_blackboard();
94 }
95 // Case 3: Check if it's a Blackboard
96 else if (py::isinstance<yasmin::Blackboard>(blackboard_obj)) {
97 blackboard = blackboard_obj.cast<Blackboard::SharedPtr>();
98 }
99 // Case 4: Unknown type - create a new blackboard
100 else {
101 blackboard = yasmin::Blackboard::make_shared();
102 }
103
104 return blackboard;
105}

119template <typename Func> inline auto wrap_blackboard_callback(py::function cb) {
120 auto holder = std::make_shared<PythonCallbackHolder>(std::move(cb));
121 return [holder](Blackboard::SharedPtr blackboard, auto... args) {
122 py::gil_scoped_acquire acquire;
123 yasmin::BlackboardPyWrapper wrapper(blackboard);
124 holder->callback()(wrapper, args...);
125 };
126}

140template <typename ReturnType>
141inline auto wrap_blackboard_callback_with_return(py::function cb) {
142 auto holder = std::make_shared<PythonCallbackHolder>(std::move(cb));
143 return [holder](Blackboard::SharedPtr blackboard) -> ReturnType {
144 py::gil_scoped_acquire acquire;
145 yasmin::BlackboardPyWrapper wrapper(blackboard);
146 return holder->callback()(wrapper).template cast<ReturnType>();

164template <typename ClassType, typename StateType>
165inline void add_call_operator(ClassType &cls) {
166 cls.def(
167 "__call__",
168 [](StateType &self, py::object blackboard_obj = py::none()) {
169 auto blackboard = convert_blackboard_from_python(blackboard_obj);
170 // Release GIL to allow C++ threads (important for Concurrence) to run
171 py::gil_scoped_release release;
172 return self(blackboard);
173 },
174 "Execute the state and return the outcome",
175 py::arg("blackboard") = py::none());
176}

185inline PyThreadState *&default_gil_saved_state() {
186 thread_local PyThreadState *state = nullptr;
187 return state;
188}
189
197inline void default_gil_before_fork() {
198 if (Py_IsInitialized() && PyGILState_Check()) {
199 default_gil_saved_state() = PyEval_SaveThread();
200 }
201}
202
208inline void default_gil_after_join() {
209 if (default_gil_saved_state()) {
210 PyEval_RestoreThread(default_gil_saved_state());
211 default_gil_saved_state() = nullptr;
212 }
213}

224template <typename Container> inline void register_default_gil_hooks() {
225 Container::set_thread_hooks(default_gil_before_fork, default_gil_after_join);
226}
