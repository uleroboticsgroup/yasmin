290 void cancel_state() override {
291
292 bool waiting_for_cancel = false;
293
294 {
295 std::lock_guard<std::mutex> lock(this->goal_handle_mutex);
296
297 // Remember the request even if the goal response has not arrived yet.
298 // The goal response callback cancels the newly accepted goal.
299 this->cancel_requested_ = true;
300
301 if (this->goal_handle && !this->action_done_.load()) {
302 try {
303 auto callback_guard = this->callback_guard;
304 this->action_client->async_cancel_goal(
305 this->goal_handle,
306 [this, callback_guard](typename rclcpp_action::Client<
307 ActionT>::CancelResponse::SharedPtr) {
308 std::lock_guard<std::mutex> guard_lock(callback_guard->mutex);
309 if (callback_guard->alive) {
310 this->cancel_done();
311 }
312 });
313 waiting_for_cancel = true;
314 } catch (const std::exception &e) {
315 YASMIN_LOG_WARN("Failed to cancel action '%s': %s",
316 this->action_name.c_str(), e.what());
317 }
318 }
319 }
320
321 // Wait for cancellation to complete outside of the goal_handle_mutex lock
322 if (waiting_for_cancel) {
323 std::unique_lock<std::mutex> cancel_lock(this->action_cancel_mutex);
324 this->action_cancel_cond.wait_for(
325 cancel_lock, std::chrono::seconds(1),
326 [this]() { return this->cancel_done_; });
327 this->cancel_done_ = false;
328 }
329
330 // Wake up the execute() method if it's waiting
331 yasmin::State::cancel_state();
332 this->action_done_cond.notify_all();
333 }

341 void cancel_done() {
342 {
343 std::lock_guard<std::mutex> lock(this->action_cancel_mutex);
344 this->cancel_done_ = true;
345 }
346 this->action_cancel_cond.notify_all();
347 }

365 std::string execute(yasmin::Blackboard::SharedPtr blackboard) override {
366
367 std::unique_lock<std::mutex> lock(this->action_done_mutex);
368 int retry_count = 0;
369
370 // Wait for the action server to be available
371 YASMIN_LOG_INFO("Waiting for action '%s'", this->action_name.c_str());
372
373 const auto action_wait_timeout =
374 std::chrono::duration<int64_t, std::ratio<1>>(this->wait_timeout);
375 const auto wait_slice = std::chrono::milliseconds(100);
376 auto waited = std::chrono::milliseconds::zero();
377
378 while (!this->action_client->wait_for_action_server(wait_slice)) {
379
380 if (this->is_canceled()) {
381 return basic_outcomes::CANCEL;
382 }
383
384 if (this->wait_timeout < 0) {
385 continue;
386 }
387
388 waited += wait_slice;
389
390 if (waited < action_wait_timeout) {
391 continue;
392 }
393
394 waited = std::chrono::milliseconds::zero();
395 YASMIN_LOG_WARN("Timeout reached, action '%s' is not available",
396 this->action_name.c_str());
397 if (retry_count < this->maximum_retry) {
398 retry_count++;
399 YASMIN_LOG_WARN("Retrying to connect to action '%s' "
400 "(%d/%d)",
401 this->action_name.c_str(), retry_count,
402 this->maximum_retry);
403 } else {
404 return basic_outcomes::TIMEOUT;
405 }
406 }
407
408 if (this->is_canceled()) {
409 return basic_outcomes::CANCEL;
410 }
411
412 Goal goal = this->create_goal_handler(blackboard);
413
414 // Prepare options for sending the goal
415 const uint64_t epoch = ++this->action_epoch_;
416 auto callback_guard = this->callback_guard;
417 SendGoalOptions send_goal_options;
418 send_goal_options.goal_response_callback =
419 [this, callback_guard, epoch](GoalResponseArg goal_response) {
420 ActionState::goal_response_callback_guarded(this, callback_guard,
421 epoch, goal_response);
422 };
423
424 send_goal_options.result_callback =
425 [this, callback_guard,
426 epoch](const typename GoalHandle::WrappedResult &result) {
427 std::lock_guard<std::mutex> lock(callback_guard->mutex);
428 if (callback_guard->alive && this->action_epoch_.load() == epoch) {
429 this->result_callback(result);
430 }
431 };
432
433 if (this->feedback_handler) {
434 // blackboard is a SharedPtr captured by value, keeping the blackboard
435 // alive for the duration of the feedback callback
436 send_goal_options.feedback_callback =
437 [this, callback_guard, epoch,
438 blackboard](typename GoalHandle::SharedPtr,
439 std::shared_ptr<const Feedback> feedback) {
440 std::lock_guard<std::mutex> lock(callback_guard->mutex);
441 if (callback_guard->alive && this->action_epoch_.load() == epoch) {
442 this->feedback_handler(blackboard, feedback);
443 }
444 };
445 }
446
447 // Send the goal to the action server. Clear any handle/flag from the
448 // previous execution so a cancel during submission applies to this goal.
449 {
450 std::lock_guard<std::mutex> lock(this->goal_handle_mutex);
451 this->goal_handle.reset();
452 this->cancel_requested_ = false;
453 }
454
455 YASMIN_LOG_INFO("Sending goal to action '%s'", this->action_name.c_str());
456 this->action_done_.store(false);
457 this->action_client->async_send_goal(goal, send_goal_options);
458
459 // Reset retry_count for the response-wait phase
460 retry_count = 0;
461
462 if (this->response_timeout > 0) {
463 // Use a single total timeout instead of a retry loop that doesn't
464 // re-send the request on each iteration
465 const auto total_timeout =
466 std::chrono::seconds(static_cast<int64_t>(this->response_timeout) *
467 (static_cast<int64_t>(this->maximum_retry) + 1));
468 if (!this->action_done_cond.wait_for(lock, total_timeout, [this]() {
469 return this->action_done_.load() || this->is_canceled();
470 })) {
471 if (this->is_canceled()) {
472 return basic_outcomes::CANCEL;
473 }
474 return basic_outcomes::TIMEOUT;
475 }
476
477 } else {
478 this->action_done_cond.wait(lock, [this]() {
479 return this->action_done_.load() || this->is_canceled();
480 });
481 }
482
483 if (this->is_canceled()) {
484 return basic_outcomes::CANCEL;
485 }
486
487 switch (this->action_status) {
488 case rclcpp_action::ResultCode::CANCELED:
489 return basic_outcomes::CANCEL;
490
491 case rclcpp_action::ResultCode::ABORTED:
492 return basic_outcomes::ABORT;
493
494 case rclcpp_action::ResultCode::SUCCEEDED:
495 if (this->result_handler) {
496 return this->result_handler(blackboard, this->action_result);
497 }
498 return basic_outcomes::SUCCEED;
499
500 default:
501 return basic_outcomes::ABORT;
502 }
503 }

575 template <typename GoalResponse>
576 static void
577 goal_response_callback_guarded(ActionState *state,
578 std::shared_ptr<CallbackGuard> callback_guard,
579 uint64_t epoch, GoalResponse goal_response) {
580 std::lock_guard<std::mutex> lock(callback_guard->mutex);
581 if (!callback_guard->alive) {
582 return;
583 }
584 if (state->action_epoch_.load() == epoch) {
585 state->goal_response_callback(goal_response);
586 } else {
587 // The state moved on to a new execution before this goal response
588 // arrived. Cancel the abandoned goal so it cannot occupy the server
589 // and make the next goal fail.
590 cancel_stale_goal(state, goal_response);
591 }
592 }

607 template <typename GoalResponse>
608 static void cancel_stale_goal(ActionState *state,
609 const GoalResponse &goal_response) {
610 const std::shared_ptr<GoalHandle> goal_handle =
611 goal_handle_from_response(goal_response);
612 if (!goal_handle) {
613 return;
614 }
615 try {
616 state->action_client->async_cancel_goal(goal_handle);
617 } catch (const std::exception &e) {
618 YASMIN_LOG_WARN("Failed to cancel stale goal for action '%s': %s",
619 state->action_name.c_str(), e.what());
620 }
621 }

633 void handle_goal_response(const std::shared_ptr<GoalHandle> &goal_handle) {
634 if (goal_handle == nullptr) {
635 std::lock_guard<std::mutex> lock(this->goal_handle_mutex);
636 this->goal_handle = nullptr;
637 this->action_status = rclcpp_action::ResultCode::ABORTED;
638 this->action_done_.store(true);
639 this->action_done_cond.notify_one();
640 return;
641 }
642
643 bool cancel = false;
644 {
645 std::lock_guard<std::mutex> lock(this->goal_handle_mutex);
646 this->goal_handle = goal_handle;
647 cancel = this->cancel_requested_ || this->is_canceled();
648 }
649
650 if (cancel) {
651 try {
652 this->action_client->async_cancel_goal(goal_handle);
653 } catch (const std::exception &e) {
654 YASMIN_LOG_WARN("Failed to cancel action '%s': %s",
655 this->action_name.c_str(), e.what());
656 }
657 }
