125template <yasmin::LogLevel LEVEL>
126void log_helper(const char *file, const char *function, int line,
127 const char *text, ...) {
128 va_list args;
129 va_start(args, text);
130
131 // Calculate the required buffer size
132 int size = vsnprintf(nullptr, 0, text, args);
133 va_end(args);
134
135 if (size < 0) {
136 yasmin::log_message.load()(LEVEL, file, function, line,
137 "[log_helper error: vsnprintf failed]");
138 return;
139 }
140
141 std::string buffer(size + 1, '\0');
142 va_start(args, text);
143 vsnprintf(&buffer[0], buffer.size(), text, args);
144
145 va_end(args);
146
147 yasmin::log_message.load()(LEVEL, file, function, line, buffer.c_str());
148}

158inline const char *extract_filename(const char *path) {
159 const char *filename = std::strrchr(path, '/');
160 if (!filename) {
161 filename = std::strrchr(path, '\\'); // handle Windows-style paths
162 }
163 return filename ? filename + 1 : path;
164}
